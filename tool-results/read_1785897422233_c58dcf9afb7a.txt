    41→
    42→function genSpark(seed: number, positive: boolean): number[] {
    43→  const arr: number[] = [];
    44→  let v = 50;
    45→  for (let i = 0; i < 18; i++) {
    46→    const noise = (Math.sin(seed + i * 1.7) + Math.cos(seed + i * 0.9)) * 6;
    47→    v += (positive ? 1.3 : -1.3) + noise * 0.45;
    48→    arr.push(Math.max(1, v));
    49→  }
    50→  return arr;
    51→}
